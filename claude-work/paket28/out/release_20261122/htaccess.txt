# mediaskills.ch - Sicherheits-Header (E124) und HTTPS-Weiterleitung (E128), Stand 19.11.2026 (Variante bei Weiterleitungsschleife laut Infomaniak FAQ 1961)
# BEGIN mediaskills HTTPS-Weiterleitung (Infomaniak FAQ 1961)
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
</IfModule>
# END mediaskills HTTPS-Weiterleitung
# BEGIN mediaskills Sicherheits-Header
<IfModule mod_headers.c>
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self'; font-src 'self'; connect-src 'self'; object-src 'none'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
# END mediaskills Sicherheits-Header
# BEGIN mediaskills Cache-Regel (E176, Paket 28)
<IfModule mod_headers.c>
  <FilesMatch "^((daten|suche)-[0-9a-f]{12}\.js|app-v[0-9]+\.js|styles-v[0-9]+\.css)$">
    Header set Cache-Control "max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "^([^/]+\.html|robots\.txt|sitemap\.xml)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>
# END mediaskills Cache-Regel
